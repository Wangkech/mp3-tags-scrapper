from MusicFile import Song, Metadata

class Folder: 
    def __init__(self, path):
        self.path =  path
        self.name = self.path.name
        self.songs = self.get_songs()

    def get_songs(self):
        files = []
        for file in self.path.iterdir():
            if file.suffix in ['.mp3','.m4a','.flac','.wav']:              
                files.append(Song(file))
        return files

class Song:

        self.__load_audio(file)
        self.__get_tags(file)


    def __load_audio(self, file):
        audio = File(file)
        self.path = file
        self.name = file.name
        self.extension= file.suffix.lower()
        self.size = file.stat().st_size /  (1024 * 1024)
        self.length =  int(audio.info.length) / 60

    def __get_tags(self,file):
        try:
            metadata = EasyID3(file)
            if metadata :
                self.metadata = Metadata(metadata)
        except: 
            self.metadata = None
